# HappyWedz
  API built with AI ML features for personalize styling

# framework, database, and other technologies used
framework - flask
database - postgresql
api testing tool - postman
environment - conda

# Installation and running the projects
clone repo from git
create conda env (conda create -n happywedzenv python=3.12)
activate conda env (conda activate happywedzenv)
install requirements.txt (pip install -r requirements.txt)
run the project (python run.py)

# versions
python - 3.12.7
psql - 17.6
conda - 25.5.1
